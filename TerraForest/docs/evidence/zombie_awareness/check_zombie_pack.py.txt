from pathlib import Path
import datetime
import hashlib
import json
import sys

root = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(root / 'tools'))
from run_godot_check import command_for, execute
from run import parse_args
from play_checkpoint import revision_metadata

test = sys.argv[1]
assert test in ('zombie_population_world','zombie_behavior')
config, _ = parse_args([])
engine = Path(config.godot).resolve()
stamp = datetime.datetime.now(datetime.timezone.utc).strftime('%Y%m%dT%H%M%S%fZ')
folder = root / 'reports' / 'checks' / (stamp + '_' + test + '_pack')
folder.mkdir(parents=True)
pack = root / 'dist' / 'TerraForest.pck'
option = '--zombie-evidence-dir=' if test == 'zombie_population_world' else '--zombie-behavior-dir='
command, cache = command_for(engine, test, folder,
    ['--gameplay-construction', option + str(folder / 'captures')])
command[command.index('--path') + 1] = str(root / 'dist')
command[command.index('--script') + 1] = str(root / 'tests' / (test+'.gd'))
command[1:1] = ['--main-pack', str(pack)]
details = dict(command=command, cache=cache, utc=stamp, timeout=100,
    required_text='failures=0',
    scope='External assertion script; res:// game resources/native libraries resolve from exported dist. Temporary world and fresh derived cache.',
    engine_sha256=hashlib.sha256(engine.read_bytes()).hexdigest(),
    pack_sha256=hashlib.sha256(pack.read_bytes()).hexdigest(),
    test_sha256=hashlib.sha256((root/'tests'/(test+'.gd')).read_bytes()).hexdigest(),
    native_sha256={str(p.relative_to(root/'dist')):hashlib.sha256(p.read_bytes()).hexdigest()
                   for p in sorted((root/'dist'/'addons').glob('*/bin/*.dll'))},
    **revision_metadata())
manifest = folder/'run.json'
manifest.write_text(json.dumps(details,indent=2)+'\n',encoding='utf-8')
details['result']=execute(command,folder,100,details['required_text'])
manifest.write_text(json.dumps(details,indent=2)+'\n',encoding='utf-8')
print(('PASS ' if details['result']['passed'] else 'FAIL ')+str(folder),flush=True)
raise SystemExit(0 if details['result']['passed'] else 1)
